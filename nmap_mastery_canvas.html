TCP/IP Fundamentals for Nmap
Three-Way Handshake
# Normal TCP connection Client → SYN → Server Client ← SYN-ACK ← Server Client → ACK → Server # SYN Scan (stealth) Client → SYN → Server Client ← SYN-ACK ← Server Client → RST → Server # Never completes
Port States
open — Service listening, SYN-ACK received filtered — Firewall drops packets (no response) closed — RST received, no service unfiltered — ACK scan; reachable, state unknown open|filtered — No response (UDP/FIN/NULL/Xmas)
Nmap Port Selection Cheatsheet
-p 22 Single port -p 22,80,443 Multiple ports -p 1-1024 Port range -p- All 65535 ports -p U:53,T:80 UDP + TCP mix --top-ports 100 Top N common ports -F Fast scan (top 100) -r Sequential port order
Output Formats
-oN scan.txt Normal output -oX scan.xml XML (parse with tools) -oG scan.gnmap Grepable output -oA basename All three formats -v / -vv Verbose / Very verbose --open Show only open ports

Fragmentation & Packet Manipulation
# Fragment packets (8 bytes) nmap -f 10.0.0.1 # Fragment (16 bytes) nmap -ff 10.0.0.1 # Custom MTU (must be mult of 8) nmap --mtu 24 10.0.0.1 # Send in custom packet sizes nmap --data-length 25 10.0.0.1 # Custom IP TTL (confuse traceroute) nmap --ttl 128 10.0.0.1 # Randomize host scan order nmap --randomize-hosts -sn 10.0.0.0/24 # Append random data nmap --data-string "GET / HTTP/1.0\r\n" 10.0.0.1
Decoy & Source Manipulation
# Decoy scan — spoof other IPs nmap -D RND:10 10.0.0.1 # Specific decoys (your IP included) nmap -D 1.2.3.4,5.6.7.8,ME 10.0.0.1 # Spoof source IP (requires root) nmap -S 1.2.3.4 -e eth0 10.0.0.1 # Use specific source port nmap --source-port 53 10.0.0.1 nmap -g 53 10.0.0.1 # Idle/Zombie scan (IP ID side channel) nmap -sI zombie_host 10.0.0.1 # Custom source port + decoy nmap -g 80 -D RND:5 10.0.0.1
Firewall Bypass Concepts
# ACK scan to map firewall rules nmap -sA 10.0.0.1 # unfiltered = port reachable past FW # filtered = FW blocking the ACK # Scan behind NAT via bounce nmap -b ftp_server 10.0.0.1 # Source port 443 to blend with HTTPS nmap -g 443 -sS 10.0.0.1 # Source port 80 — blend with HTTP nmap --source-port 80 10.0.0.1 # IPv6 scanning (less monitored) nmap -6 2001:db8::1 # Slow scan to avoid rate-based IDS nmap -T0 --scan-delay 30s 10.0.0.1
Proxy & Timing Evasion

Bash Automation Script — Full Recon Pipeline
#!/bin/bash # nmap_recon.sh — Automated pentest scanner # Usage: ./nmap_recon.sh 10.0.0.0/24 /output/dir TARGET="$1" OUTPUT="${2:-./nmap_output}" TIMESTAMP=$(date +%Y%m%d_%H%M%S) DIR="${OUTPUT}/${TIMESTAMP}" mkdir -p "$DIR" echo "[*] Starting scan: $TARGET → $DIR" # Phase 1: Host discovery echo "[*] Phase 1: Host discovery..." nmap -sn -T4 --min-rate 2000 -PE -PP -PS22,80,443 \ "$TARGET" -oG "$DIR/01_alive.gnmap" 2>&1 ALIVE=$(grep "Up" "$DIR/01_alive.gnmap" | awk '{print $2}') echo "$ALIVE" > "$DIR/alive_hosts.txt" COUNT=$(echo "$ALIVE" | wc -l) echo "[+] Found $COUNT alive hosts" # Phase 2: Port scan echo "[*] Phase 2: Full port scan..." nmap -sS -p- -T4 --min-rate 5000 --max-retries 2 \ -iL "$DIR/alive_hosts.txt" \ -oA "$DIR/02_ports" 2>&1 # Phase 3: Service detection echo "[*] Phase 3: Service enumeration..." nmap -sV -sC --open -iL "$DIR/alive_hosts.txt" \ -oA "$DIR/03_services" 2>&1 # Phase 4: Vuln scan echo "[*] Phase 4: Vulnerability scan..." nmap --script vuln --open -iL "$DIR/alive_hosts.txt" \ -oA "$DIR/04_vulns" 2>&1 # Generate summary echo "[+] Scan complete. Summary:" grep "open" "$DIR/02_ports.gnmap" | \ awk '{print $2, $5}' | sort | uniq -c | sort -rn | head -20 echo "[+] Reports saved to: $DIR"
Python Automation — Nmap with python-nmap
#!/usr/bin/env python3 # nmap_auto.py — Python-based scan orchestrator import nmap, json, sys, datetime from pathlib import Path class NmapScanner: def __init__(self, target, output_dir="./results"): self.nm = nmap.PortScanner() self.target = target self.output = Path(output_dir) self.output.mkdir(exist_ok=True) self.findings = [] def discover_hosts(self): print(f"[*] Discovering hosts: {self.target}") self.nm.scan(hosts=self.target, arguments="-sn -T4") return [h for h in self.nm.all_hosts() if self.nm[h].state() == 'up'] def port_scan(self, hosts): print(f"[*] Scanning {len(hosts)} hosts...") targets = " ".join(hosts) self.nm.scan(hosts=targets, arguments="-sS -sV -sC --open -T4 -p 1-65535") for host in self.nm.all_hosts(): for proto in self.nm[host].all_protocols(): ports = self.nm[host][proto].keys() for port in ports: info = self.nm[host][proto][port] if info['state'] == 'open': self.findings.append({ 'host': host, 'port': port, 'proto': proto, 'service': info.get('name', ''), 'version': info.get('product', ''), 'script': info.get('script', {}) }) def report(self): ts = datetime.datetime.now().strftime("%Y%m%d_%H%M%S") out = self.output / f"scan_{ts}.json" with open(out, 'w') as f: json.dump(self.findings, f, indent=2) print(f"[+] {len(self.findings)} open ports found") print(f"[+] Report: {out}") return self.findings if __name__ == "__main__": scanner = NmapScanner(sys.argv[1]) hosts = scanner.discover_hosts() scanner.port_scan(hosts) scanner.report()

Nmap XML Parser (Python)
import xml.etree.ElementTree as ET def parse_nmap(xml_file): tree = ET.parse(xml_file) root = tree.getroot() findings = [] for host in root.findall('host'): ip = host.find( 'address[@addrtype="ipv4"]' ).get('addr') for port in host.findall( './/port'): state = port.find('state') if state.get('state') == 'open': svc = port.find('service') findings.append({ 'ip': ip, 'port': port.get('portid'), 'service': svc.get('name') }) return findings
