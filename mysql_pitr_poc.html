STEP 05
Take Full Backup (The Safety Net)
mysqldump with --master-data=2 — records the binlog position
📸
This is the "2 AM backup" in your real scenario. In the POC, you're taking it now after inserting the 1000 rows. After the backup, insert a few MORE rows to simulate "morning activity".
terminal — full backup with PITR flags
BACKUP
# Take full backup with --master-data=2 (records binlog position)
$ docker exec mysql-pitr \
  mysqldump \
  -u root -ppitrpoc123 \
  --all-databases \
  --single-transaction \
  --master-data=2 \
  --flush-logs \
  --events --routines \
  > ~/mysql-pitr-poc/backups/full_backup.sql

STEP 07
⏪ Run PITR Recovery — Get Data Back
3 sub-steps: copy binlogs → restore backup → replay binlogs to safe time
7A ─ COPY BINLOGS OUT OF CONTAINER
Like taking the DVR tapes out of the server rack.
7A — copy binlogs from container
STEP 7A
# See what binlog files exist inside the container
$ docker exec mysql-pitr \
  ls -lth /var/lib/mysql/mysql-bin.*
-rw-r----- mysql-bin.000003  4.0K  (new clean file after FLUSH)
-rw-r----- mysql-bin.000002  512K  ← this has the DELETE event
-rw-r----- mysql-bin.000001  1.0M  ← initial binlog

# Copy ALL binlog files to your local binlogs folder
$ docker cp \
  mysql-pitr:/var/lib/mysql/mysql-bin.000001 \
  ~/mysql-pitr-poc/binlogs/

$ docker cp \
  mysql-pitr:/var/lib/mysql/mysql-bin.000002 \
  ~/mysql-pitr-poc/binlogs/

$ docker cp \
  mysql-pitr:/var/lib/mysql/mysql-bin.000003 \
  ~/mysql-pitr-poc/binlogs/

# Verify all copied
$ ls -lh ~/mysql-pitr-poc/binlogs/
-rw-r--r-- mysql-bin.000001  1.0M
-rw-r--r-- mysql-bin.000002  512K
-rw-r--r-- mysql-bin.000003  4.0K  ✔ all 3 files copied
7B ─ FIND THE DELETE POSITION IN BINLOG
Read the binlog to find the exact position of the DELETE event.
7B — find the DELETE position
STEP 7B
# Read binlog 000002 — find the DELETE event
$ mysqlbinlog \
  --base64-output=DECODE-ROWS \
  --verbose \
  ~/mysql-pitr-poc/binlogs/mysql-bin.000002 \
  | grep -B10 "DELETE FROM"

# OUTPUT YOU'LL SEE:
# # at 4891           ← ★ START POSITION of DELETE (write this down!)
# #241201 11:00:25 server id 1  end_log_pos 4960
# ### DELETE FROM `shopdb`.`orders`
#        ↑ found it!

# Alternative: use --stop-datetime approach (easier, slightly less precise)
# If your safe_time from Step 5 was "2024-12-01 10:59:55",
# use that as the --stop-datetime in the next step.

# Show all events with timestamps for 000002
$ mysqlbinlog --base64-output=DECODE-ROWS --verbose \
  ~/mysql-pitr-poc/binlogs/mysql-bin.000002 \
  | grep -E "^# at |DELETE|INSERT"
7C ─ RESTORE BACKUP + REPLAY BINLOGS (THE MAIN EVENT)
This is the actual recovery. Two sub-commands: restore dump → replay logs up to safe time.
7C — THE ACTUAL RECOVERY COMMANDS
⭐ KEY STEP
# ── OPTION A: Recover to safe timestamp (simpler) ─────────────────
# Replace 10:59:55 with YOUR actual safe_time from Step 5

# First: restore the full backup (puts DB back at backup state)
$ docker exec -i mysql-pitr \
  mysql -u root -ppitrpoc123 \
  < ~/mysql-pitr-poc/backups/full_backup.sql

# Verify: backup restore worked (should show 1000 rows)
$ docker exec mysql-pitr \
  mysql -u root -ppitrpoc123 shopdb -e "SELECT COUNT(*) FROM orders;"
+----------+
| COUNT(*) |
+----------+
|     1000 |   ← backup restored (1000 rows, morning 3 not yet applied)
+----------+

# Second: replay binlogs UP TO safe time (adds the morning 3 rows back)
# ↓ THE MAGIC COMMAND ↓
$ mysqlbinlog \
  --start-datetime="2024-12-01 00:00:00" \
  --stop-datetime="2024-12-01 10:59:55" \
  ~/mysql-pitr-poc/binlogs/mysql-bin.000001 \
  ~/mysql-pitr-poc/binlogs/mysql-bin.000002 \
  | docker exec -i mysql-pitr \
  mysql -u root -ppitrpoc123

# ── OPTION B: Recover using exact position (more precise) ─────────
# Use the position number from 7B (e.g., 4890 = one before DELETE at 4891)

$ mysqlbinlog \
  --stop-position=4890 \
  ~/mysql-pitr-poc/binlogs/mysql-bin.000002 \
  | docker exec -i mysql-pitr \
  mysql -u root -ppitrpoc123

⚡ Complete POC in 9 Commands (Cheat Summary)
complete poc — all 9 key commands
FULL SCRIPT
# 1. SETUP
mkdir -p ~/mysql-pitr-poc/{backups,binlogs,config}
echo -e "[mysqld]\nserver-id=1\nlog_bin=/var/lib/mysql/mysql-bin\nbinlog_format=ROW" > ~/mysql-pitr-poc/config/my.cnf

# 2. START MYSQL
docker run -d --name mysql-pitr -e MYSQL_ROOT_PASSWORD=pitrpoc123 \
  -e MYSQL_DATABASE=shopdb -p 3306:3306 \
  -v ~/mysql-pitr-poc/config/my.cnf:/etc/mysql/conf.d/my.cnf \
  -v ~/mysql-pitr-poc/backups:/backup mysql:8.0 && sleep 25

# 3. INSERT 1000 ROWS
docker exec mysql-pitr mysql -u root -ppitrpoc123 shopdb -e \
  "CREATE TABLE orders(id INT AUTO_INCREMENT PRIMARY KEY, customer VARCHAR(100), product VARCHAR(100), amount DECIMAL(10,2), created_at DATETIME DEFAULT NOW()); INSERT INTO orders(customer,product,amount) SELECT CONCAT('C_',seq), CONCAT('P_',seq%50), ROUND(RAND()*500,2) FROM (SELECT a.N+b.N*10+c.N*100+1 AS seq FROM (SELECT 0 N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) a, (SELECT 0 N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) b, (SELECT 0 N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) c) t WHERE seq<=1000;"

# 4. BACKUP
docker exec mysql-pitr mysqldump -u root -ppitrpoc123 --all-databases \
  --single-transaction --master-data=2 --flush-logs \
  > ~/mysql-pitr-poc/backups/full_backup.sql

# 5. INSERT MORE ROWS + NOTE SAFE TIME
docker exec mysql-pitr mysql -u root -ppitrpoc123 shopdb -e \
  "INSERT INTO orders(customer,product,amount) VALUES('Morning_1','ProdA',99.99),('Morning_2','ProdB',149.50);"
SAFE_TIME=$(docker exec mysql-pitr mysql -u root -ppitrpoc123 -sNe "SELECT DATE_FORMAT(NOW(),'%Y-%m-%d %H:%i:%s');")
echo "Safe time: $SAFE_TIME"

# 6. 💥 DISASTER
sleep 3
docker exec mysql-pitr mysql -u root -ppitrpoc123 shopdb -e "DELETE FROM orders;"
docker exec mysql-pitr mysql -u root -ppitrpoc123 -e "FLUSH BINARY LOGS;"
echo "Rows after disaster: $(docker exec mysql-pitr mysql -u root -ppitrpoc123 shopdb -sNe 'SELECT COUNT(*) FROM orders;')"

# 7. COPY BINLOGS
for f in 000001 000002 000003; do
  docker cp mysql-pitr:/var/lib/mysql/mysql-bin.$f ~/mysql-pitr-poc/binlogs/ 2>/dev/null
done

# 8. ⏪ PITR RECOVERY
docker exec -i mysql-pitr mysql -u root -ppitrpoc123 < ~/mysql-pitr-poc/backups/full_backup.sql
mysqlbinlog --stop-datetime="$SAFE_TIME" \
  ~/mysql-pitr-poc/binlogs/mysql-bin.000001 \
  ~/mysql-pitr-poc/binlogs/mysql-bin.000002 \
  | docker exec -i mysql-pitr mysql -u root -ppitrpoc123

# 9. ✅ VERIFY
docker exec mysql-pitr mysql -u root -ppitrpoc123 shopdb -e "SELECT COUNT(*) AS recovered_rows FROM orders;"
