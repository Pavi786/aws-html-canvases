Core flags

-o          Output file
-oJ         JSON output

DNS takeover detection
DNS takeover occurs when a CNAME points to an unclaimed third-party service. High severity in bug bounty — typically $500–$5000.

Bash automation — full recon pipeline
#!/bin/bash
# recon.sh — Full passive recon pipeline
# Usage: ./recon.sh target.com /output/dir

TARGET="$1"
OUT="${2:-./recon_$(date +%Y%m%d_%H%M%S)}"
mkdir -p "$OUT"

echo "[*] Starting recon: $TARGET"

# Phase 1: Subdomain enumeration
echo "[*] Phase 1: Passive subdomain enum..."
subfinder -d $TARGET -all -silent -o "$OUT/subs.txt"
amass enum -passive -d $TARGET >> "$OUT/subs.txt"
cat "$OUT/subs.txt" | sort -u -o "$OUT/subs.txt"
echo "[+] $(wc -l < $OUT/subs.txt) unique subdomains"

# Phase 2: DNS resolution
echo "[*] Phase 2: DNS resolution..."
dnsx -l "$OUT/subs.txt" -silent -a -cname -resp \
  -o "$OUT/resolved.txt"

# Phase 3: HTTP probing
echo "[*] Phase 3: HTTP probing..."
cat "$OUT/subs.txt" | httpx \
  -silent -mc 200,301,302,401,403 \
  -title -tech-detect -status-code \
  -o "$OUT/live.txt"

# Phase 4: Vulnerability scan
echo "[*] Phase 4: Nuclei vuln scan..."
nuclei -l "$OUT/live.txt" \
  -t cves/,vulnerabilities/,exposures/ \
  -severity critical,high,medium \
  -rate-limit 100 \
  -o "$OUT/vulns.txt"

# Phase 5: DNS takeover check
echo "[*] Phase 5: Takeover check..."
subjack -w "$OUT/subs.txt" \
  -t 100 -o "$OUT/takeovers.txt" -ssl

# Summary
echo ""
echo "=== RECON COMPLETE ==="
echo "Subdomains: $(wc -l < $OUT/subs.txt)"
echo "Live hosts:  $(wc -l < $OUT/live.txt)"
echo "Vulns found: $(wc -l < $OUT/vulns.txt)"
echo "Output:      $OUT"
Python automation — recon orchestrator
#!/usr/bin/env python3
# recon_orchestrator.py
import subprocess, json, sys, datetime
from pathlib import Path

class ReconPipeline:
    def __init__(self, domain, output_dir="./recon"):
        self.domain = domain
        self.out = Path(output_dir) / domain / datetime.datetime.now().strftime("%Y%m%d")
        self.out.mkdir(parents=True, exist_ok=True)
        self.results = {}

    def run(self, cmd, desc):
        print(f"[*] {desc}...")
        result = subprocess.run(
            cmd, shell=True, capture_output=True, text=True
        )
        return result.stdout.strip().splitlines()

    def subfinder(self):
        out = self.out / "subs.txt"
        subs = self.run(
            f"subfinder -d {self.domain} -all -silent",
            "Running subfinder"
        )
        out.write_text("\n".join(subs))
        self.results['subdomains'] = len(subs)
        return subs

    def httpx_probe(self, subs):
        inp = self.out / "subs.txt"
        out = self.out / "live.json"
        self.run(
            f"cat {inp} | httpx -silent -json -o {out} -mc 200,301,302,401,403",
            "Probing live hosts"
        )
        try:
            data = json.loads(f"[{out.read_text().strip().replace(chr(10),',')}]")
            self.results['live'] = len(data)
            return data
        except: return []

    def nuclei_scan(self):
        live = self.out / "live.json"
        out = self.out / "vulns.txt"
        self.run(
            f"nuclei -l {live} -severity critical,high -o {out} -silent",
            "Running nuclei scan"
        )
        vulns = out.read_text().splitlines() if out.exists() else []
        self.results['vulns'] = len(vulns)
        return vulns

    def report(self):
        report = self.out / "summary.json"
        report.write_text(json.dumps(self.results, indent=2))
        print(f"\n[+] Recon complete: {self.domain}")
        print(json.dumps(self.results, indent=2))

    def run_all(self):
        subs = self.subfinder()
        live = self.httpx_probe(subs)
        vulns = self.nuclei_scan()
        self.report()

if __name__ == "__main__":
    p = ReconPipeline(sys.argv[1])
    p.run_all()
Continuous monitoring — new subdomain alerts
#!/bin/bash
# monitor.sh — Runs daily via cron, alerts on new subs
TARGET="$1"
DB="~/.recon_db/$TARGET"
mkdir -p $DB
TODAY="$DB/$(date +%Y%m%d).txt"
PREV="$DB/latest.txt"
SLACK_WEBHOOK="https://hooks.slack.com/services/XXX"

# Run current scan
subfinder -d $TARGET -all -silent | sort -u > $TODAY

# Compare with previous run
if [ -f "$PREV" ]; then
  NEW=$(comm -13 $PREV $TODAY)
  if [ -n "$NEW" ]; then
    MSG="*New subdomains for $TARGET:*\n\`\`\`$NEW\`\`\`"
    curl -s -X POST $SLACK_WEBHOOK \
      -H 'Content-type: application/json' \
      -d "{\"text\":\"$MSG\"}"
  fi
fi

cp $TODAY $PREV
echo "[+] Monitoring complete: $(wc -l < $TODAY) subs"

GitHub Actions CI/CD recon automation
# .github/workflows/recon.yml
name: Continuous Recon
on:
  schedule:
    - cron: '0 6 * * *'
  workflow_dispatch:

jobs:
  recon:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3

      - name: Install tools
        run: |
          go install github.com/projectdiscovery/subfinder/v2/cmd/subfinder@latest
          go install github.com/projectdiscovery/httpx/cmd/httpx@latest
          go install github.com/projectdiscovery/nuclei/v3/cmd/nuclei@latest
          nuclei -update-templates

      - name: Configure subfinder
        run: |
          mkdir -p ~/.config/subfinder
          echo "${{ secrets.PROVIDER_CONFIG }}" > ~/.config/subfinder/provider-config.yaml

      - name: Run recon pipeline
        run: |
          subfinder -d ${{ vars.TARGET_DOMAIN }} -all -silent -o subs.txt
          cat subs.txt | httpx -silent -o live.txt -mc 200,301,302,403
          nuclei -l live.txt -severity critical,high -o vulns.txt -silent

      - name: Check for new subdomains
        run: |
          git diff --name-only HEAD~1 HEAD | grep subs.txt | \
          xargs -I{} comm -13 <(git show HEAD~1:{}) <(cat {}) > new_subs.txt
          cat new_subs.txt

      - name: Upload results
        uses: actions/upload-artifact@v3
        with:
          name: recon-results
          path: |
            subs.txt
            live.txt
            vulns.txt

      - name: Notify Slack on findings
        if: ${{ hashFiles('vulns.txt') != '' }}
        run: |
          curl -s -X POST ${{ secrets.SLACK_WEBHOOK }} \
            -H 'Content-type: application/json' \
            -d "{\"text\":\"Vulnerabilities found! Check artifacts.\"}"
